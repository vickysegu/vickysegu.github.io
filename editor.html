async function lookupSinta(judul){ if(!judul) return null; // Ambil 8 kata pertama judul sebagai query const query = judul.split(/\s+/).slice(0,8).join(' '); // ✅ DOMAIN BARU (Kemdiktisaintek) const sintaUrl = `https://sinta.kemdiktisaintek.go.id/journals?q=${encodeURIComponent(query)}`; const proxies = [ `https://api.allorigins.win/raw?url=${encodeURIComponent(sintaUrl)}`, `https://corsproxy.io/?${encodeURIComponent(sintaUrl)}`, `https://api.codetabs.com/v1/proxy?quest=${encodeURIComponent(sintaUrl)}` ]; for(const proxy of proxies){ try{ const r = await fetch(proxy); if(!r.ok) continue; const html = await r.text(); // Pola 1: "Sinta 4" / "SINTA 2" let m = html.match(/S[Ii][Nn][Tt][Aa]\s*([1-6])/); if(m) return `SINTA ${m[1]}`; // Pola 2: "Accred : Sinta 3" (format di halaman profil) m = html.match(/Accred\s*:\s*Sinta\s*([1-6])/i); if(m) return `SINTA ${m[1]}`; // Pola 3: "Current Accreditation" diikuti "Sinta X" m = html.match(/Current\s+Ac?creditation[\s\S]{0,200}?Sinta\s*([1-6])/i); if(m) return `SINTA ${m[1]}`; }catch(e){ console.warn('Proxy gagal:', e.message); } } return null; }
